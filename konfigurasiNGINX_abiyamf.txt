# Konfigurasi NGINX Reverse Proxy dan Rate Limiting
# Lokasi Berkas: /etc/nginx/sites-available/default
# Dicoding - Belajar Jaringan Komputer untuk Pemula
# Username: abiyamf

# Penerapan Rate Limit: 6 request per menit (1 request setiap 10 detik)
limit_req_zone $binary_remote_addr zone=one:10m rate=6r/m;

server {
	# Mengubah port default NGINX dari 80 ke 3000
	listen 3000 default_server;
	listen [::]:3000 default_server;

	root /var/www/html;
	index index.html index.htm index.nginx-debian.html;

	server_name _;

	location / {
		# Menerapkan pembatasan akses (rate limit)
		limit_req zone=one;

		# Mengonfigurasi Reverse Proxy ke Web Server Node.js (Port 8000)
		proxy_pass http://localhost:8000;
		proxy_http_version 1.1;
		proxy_set_header Upgrade $http_upgrade;
		proxy_set_header Connection 'upgrade';
		proxy_set_header Host $host;
		proxy_cache_bypass $http_upgrade;
	}
}
